  \node[node, text width=6.6cm, minimum height=1.55cm] (dec) at (12.4, -3.85) {%
    \textbf{Decrypt and decide}\\[1pt]
    \pipesub{margin against the frozen per-label threshold;\\ no decrypted value returns to the server}};

  \draw[lane, draw=pipeBlue, dashed, fill=pipeBlueLt] (0, -6.0) rectangle (16.2, -9.25);

  \node[lanelab] at (0, -9.65) {Evaluation};
